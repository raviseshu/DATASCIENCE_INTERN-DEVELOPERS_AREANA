FROM python:3.11-slim

WORKDIR /app

# Install dependencies
COPY requirements.txt .
RUN pip install --no-cache-dir -r requirements.txt

# Copy project files
COPY . .

# Create directories
RUN mkdir -p models plots data docs

# Expose FastAPI port
EXPOSE 8000

# Production server
CMD ["uvicorn", "api:app", "--host", "0.0.0.0", "--port", "8000"]
